cmake_minimum_required(VERSION 3.20)
project(lvgl_embedded_dashboard LANGUAGES C CXX)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
include(CTest)

option(DASHBOARD_ENABLE_SDL "Build the interactive SDL2 window" OFF)
set(LV_CONF_SKIP ON CACHE BOOL "Use LVGL defaults with explicit feature definitions" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "No upstream demos" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "No upstream examples" FORCE)
set(CONFIG_LV_USE_THORVG OFF CACHE BOOL "No vector engine needed" FORCE)

if(DASHBOARD_ENABLE_SDL)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
endif()

# Official LVGL v9.6.0 is a Git submodule pinned to
# 80ca777e37a2b176770726a02e07a6fb79ef0b39.
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/lvgl/CMakeLists.txt")
  message(FATAL_ERROR "LVGL submodule missing. Run: git submodule update --init --recursive")
endif()
add_subdirectory(third_party/lvgl)
target_compile_definitions(lvgl PUBLIC LV_USE_SDL=$<BOOL:${DASHBOARD_ENABLE_SDL}> LV_USE_THORVG=0)
if(DASHBOARD_ENABLE_SDL)
  target_link_libraries(lvgl PUBLIC PkgConfig::SDL2)
endif()

add_library(dashboard_model app/dashboard_model.c)
target_include_directories(dashboard_model PUBLIC include)
add_library(dashboard_ui app/dashboard_ui.c)
target_include_directories(dashboard_ui PUBLIC include)
target_link_libraries(dashboard_ui PUBLIC dashboard_model lvgl)

add_executable(test_model tests/test_model.c)
target_link_libraries(test_model PRIVATE dashboard_model)
add_executable(dashboard_headless tests/dashboard_headless.c)
target_link_libraries(dashboard_headless PRIVATE dashboard_ui)
if(DASHBOARD_ENABLE_SDL)
  add_executable(dashboard_demo app/dashboard_demo.c)
  target_link_libraries(dashboard_demo PRIVATE dashboard_ui)
endif()

if(BUILD_TESTING)
  add_test(NAME model COMMAND test_model)
  foreach(scenario normal fault interaction)
    add_test(NAME ${scenario} COMMAND dashboard_headless ${scenario})
    set_tests_properties(${scenario} PROPERTIES TIMEOUT 15)
  endforeach()
  if(DASHBOARD_ENABLE_SDL)
    add_test(NAME sdl_smoke COMMAND dashboard_demo --smoke)
    set_tests_properties(sdl_smoke PROPERTIES TIMEOUT 20)
  endif()
endif()
